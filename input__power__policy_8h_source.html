132inline bool can_use_pd_full_power(const SourceStatus& s)
133{
134 return s.negotiatedVoltage_mV > 0 and s.isSinkReady and
135 // the voltage/current rise should be over
136 s.vbusVoltage_mV >= pd_min_established_vbus_mV(s.negotiatedVoltage_mV);

140inline bool is_standard_port(const SourceStatus& s)
141{
142 return s.isPowerSourceDetected and
143 // let some time pass after a new detection
144 s.timeSinceSourceDetected_ms > standardPortDetectionDelay_ms and
145 // last pd detected was some time ago
146 s.timeSinceLastPdDetected_ms > pdLostDelay_ms and
147 // a late PD negociation (soft/hard reset when the source caps were missed) can take seconds.
148 // Pulling current from a fast charger before it ends can make it crash (issue #633)
149 (s.timeSinceSourceDetected_ms > maxPdNegociationWait_ms or not s.isSinkNegotiating);
150}

153inline bool can_use_power(const SourceStatus& s)
154{
155 return s.isPowerSourceDetected and ((s.isUsbPd and can_use_pd_full_power(s)) or is_standard_port(s));
156}

159inline uint16_t max_input_current_mA(const SourceStatus& s)
160{
161 // safety
162 if (not s.isPowerSourceDetected)
163 return 0;
164
165 // power delivery detected
166 if (s.isUsbPd)
167 {
168 if (can_use_pd_full_power(s))
169 return static_cast<uint16_t>(s.negotiatedCurrent_mA * pdCurrentRatio);
170 // no contract yet
171 return 0;
172 }
173 // no usb pd since some time, and vbus seems stable so try to use it
174 if (is_standard_port(s))
175 // maximum USB current (hacky, would prefer USB type detection)
176 return standardPortCurrent_mA;
177
178 // we dont known for now the type of connection
179 return 0;
180}
