209 bool set_state_locked(const State s, const bool forceUpdate)
210 {
211 if (not forceUpdate and s == current)
212 {
213 // ignore a set state with same value
214 return false;
215 }
216 // update the state
217 lastState = current;
218 current = s;
219 // reset the timeout time
220 isTimeoutSet = false;
221 timeoutStart_ms = 0;
222 timeoutDuration_ms = 0;
223 stateSetTime = hal::time_ms();
224 ++stateGeneration;
225
226 changedWithTimeout = false;
227 didStateJustChanged = true;
228 return true;
229 }

232 bool set_state_with_timeout_locked(const State s, const uint32_t timeout, const State stateOnTimeout)
233 {
234 // set the new state if
235 // - no timeout is active
236 // - new state changes
237 // - timeout state changes
238 if (not isTimeoutSet or (s != current or afterTimeoutState != stateOnTimeout))
239 {
240 // force update state
241 set_state_locked(s, true);
242 // timeout target and delay first, the timeout is active once all are set
243 afterTimeoutState = stateOnTimeout;
244 timeoutStart_ms = hal::time_ms();
245 timeoutDuration_ms = timeout;
246 isTimeoutSet = true;
247 return true;
248 }
249 return false;
