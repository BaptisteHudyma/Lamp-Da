213bool button_hold(const uint8_t clicks, const bool isEndOfHoldEvent, const uint32_t holdDuration)
214{
215 auto manager = get_context();
216 auto& rampHandler = manager.state.rampHandler;
217
218 switch (clicks)
219 {
220 case 3: // 3 click+hold: configure custom ramp
221 // no ramps in favorite group
222 if (not manager.state.isInFavoriteMockGroup)
223 {
224 rampHandler.update_ramp(manager.get_active_custom_ramp(), holdDuration, [&](uint8_t rampValue) {
225 manager.custom_ramp_update(rampValue);
226 manager.set_active_custom_ramp(rampValue);
227 });
228 return true;
229 }
230 break;
231
232 // 5 click+hold: Add 5 minutes to sunset timer
233 case 5:
234 {
235 if (not isEndOfHoldEvent and holdDuration > 0 and logic::sunset::is_enabled())
236 {
237 // sunset timer !
238 auto manager = get_context();
239 if (manager.overlay_animate_ramp(
240 holdDuration, 1000, modes::colors::PaletteGradient<modes::colors::White, modes::colors::Red>))
241 {
242 // update the sunset timing
243 manager.state.isSunsetTimingPending = 2;
244 }
245 }
246 break;
247 }
248
249 case 13: // 13 clicks + hold: reset the whole system and stored parameters
250 {
251 if (not isEndOfHoldEvent and holdDuration > 0)
252 {
253 auto manager = get_context();
254 if (manager.overlay_animate_ramp(
255 holdDuration, 5000, modes::colors::PaletteGradient<modes::colors::Red, modes::colors::Red>))
256 {
257 // reset the file system and memory
258 bsp::lampda_print("clearing the whole file format");
259 component::fileSystem::clear_internal_fs();
260
261 // shutdown the lamp
262 const bool shouldSaveUserParameters = false;
263 logic::behavior::internal::handle_shutdown_state(shouldSaveUserParameters);
264 }
265 }
266 break;
267 }
268
269 case 20: // 20 clicks + hold: reset the whole system and stored parameters
270 {
271 if (not isEndOfHoldEvent and holdDuration > 0)
272 {
273 auto manager = get_context();
274 if (manager.overlay_animate_ramp(
275 holdDuration, 5000, modes::colors::PaletteGradient<modes::colors::Red, modes::colors::Red>))
276 {
277 // reset the file system and memory
278 bsp::lampda_print("clearing the whole file format");
279 component::fileSystem::clear_internal_fs();
280
281 // shutdown the lamp
282 const bool shouldSaveUserParameters = false;
283 const bool shouldSaveSystemParameters = false;
284 logic::behavior::internal::handle_shutdown_state(shouldSaveUserParameters, shouldSaveSystemParameters);
285 }
286 }
287 break;
288 }
289 }
290
291 return false;
292}

297void handle_brightness_control(const brightness_t requiredbrightness)
298{
299 // ignore if not on
300 if (not logic::behavior::is_in_output_state())
301 return;
302
303 logic::sunset::lock_brightness_update(true);
304 // update brightness
305 const brightness_t desiredBrightness =
306 min<brightness_t>(::lampda::brightness::absoluteMaximumBrightness, requiredbrightness);
307 // Update the system brightness for real, not the temporary. We want the changes to be saved
308 logic::brightness::update_brightness(desiredBrightness);
309 // update saved brightness
310 logic::brightness::update_saved_brightness();
311 logic::sunset::lock_brightness_update(false);
312
313 // and change the sunset timer if needed
314 logic::sunset::bump_timer();
315}

374void handle_sunset_to_time_command(const component::time::RealTime& time)
375{
376 if (not time.is_valid())
377 {
378 bsp::lampda_print("Refusing sunset command %d %dh %dm %ds. Invalid values.",
379 time.dayOfTheWeek,
380 time.hour,
381 time.minutes,
382 time.seconds);
383 return;
384 }
385 const auto& realTime = component::time::get_real_time();
386 if (not realTime.is_valid())
387 {
388 bsp::lampda_print("Refusing sunset command %d %dh %dm %ds : Time is not synchronized yet.",
389 time.dayOfTheWeek,
390 time.hour,
391 time.minutes,
392 time.seconds);
393 return;
394 }
395
396 const uint32_t internalLampActionTime = component::time::get_platform_time_from_target_time(time);
397 if (internalLampActionTime <= 0)
398 {
399 bsp::lampda_print("Refusing sunset command %d %dh %dm %ds: Target time is incoherent",
400 time.dayOfTheWeek,
401 time.hour,
402 time.minutes,
403 time.seconds);
404 return;
405 }
406
407 // lamp will turn on if not already turned on
408 if (not logic::behavior::is_in_output_state())
409 logic::behavior::set_power_on();
410
411 logic::sunset::set_deadline(internalLampActionTime);
412}

440bool handle_user_command(const common::UserCommand& command)
441{
442 switch (command.get_type())
443 {
444 case common::UserCommand::Type::SetUserRamp:
445 {
446 uint8_t ramp;
447 if (command.parse_ramp(ramp))
448 __private::handle_ramp_command(ramp);
449 else
450 bsp::lampda_print("Failed to parse set_user_ramp command");
451 return true;
452 }
453 case common::UserCommand::Type::Brightness:
454 {
455 brightness_t brgt;
456 if (command.parse_brightness(brgt))
457 __private::handle_brightness_control(brgt);
458 else
459 bsp::lampda_print("Failed to parse brightness command");
460 return true;
461 }
462 case common::UserCommand::Type::SetMode:
463 {
464 uint8_t groupId;
465 uint8_t modeId;
466 if (command.parse_set_mode(groupId, modeId))
467 __private::handle_mode_control(groupId, modeId);
468 else
469 bsp::lampda_print("Failed to parse set_mode command");
470 return true;
471 }
472 case common::UserCommand::Type::OnOff:
473 {
474 bool shouldTurnOn;
475 if (command.parse_turn_onoff(shouldTurnOn))
476 __private::handle_on_off_command(shouldTurnOn);
477 else
478 bsp::lampda_print("Failed to parse onoff command");
479 return true;
480 }
481 case common::UserCommand::Type::SetRealTime:
482 {
483 component::time::RealTime time;
484 if (command.parse_set_real_time_command(time))
485 __private::handle_set_time_command(time);
486 else
487 bsp::lampda_print("Failed to parse set_real_time command");
488 return true;
489 }
490 case common::UserCommand::Type::SetSunsetToTime:
491 {
492 component::time::RealTime time;
493 if (command.parse_set_sunset_to_time_command(time))
494 __private::handle_sunset_to_time_command(time);
495 else
496 bsp::lampda_print("Failed to parse set_sunset_time command");
497 return true;
498 }
499
500 default:
501 break;
502 }
503 return false;
504}
