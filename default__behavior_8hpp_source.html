64void brightness_update(const brightness_t brightness)
65{
66 auto manager = get_context();
67
68 // dont handle invalid commands
69 if (brightness <= ::lampda::brightness::absoluteMaximumBrightness)
70 {
71 // force update of the internal references to the new brightness
72 // (the system saved brightness is only updated after this callback)
73 manager.lamp.align_internal_to_system_brightness(brightness);
74
75 // set brightness for underlying object (w/o re-entry in update_brightness)
76 manager.lamp.setBrightness(brightness, true, true);
77
78 // callbacks
79 manager.brightness_update(brightness);
80 }
81 else
82 {
83 // this call could be just a max brightness update
84 manager.lamp.enforce_internal_brightness_limits();
85 }

102bool button_start_hold_default(const uint8_t clicks, const bool isEndOfHoldEvent, const uint32_t holdDuration)
103{
104 switch (clicks)
105 {
106 case 1:
107 {
108 // 1+hold (2s): activate bluetooth advertising
109 // activate bluetooth !
110 auto manager = get_context();
111
112 // ramp can activate bluetooth :
113 // - if it is not enabled yet
114 // - if it is enabled, but the bounded peer already exists (force open connection)
115
116 // Start pairing mode
117 const bool hasPairedDeviceAndNoPairing =
118 bsp::ble::is_bounded() and not bsp::ble::is_advertising() and not bsp::ble::is_connected();
119 if (hasPairedDeviceAndNoPairing)
120 {
121 if (manager.overlay_animate_ramp(
122 holdDuration, 1000, modes::colors::PaletteGradient<modes::colors::Blue, modes::colors::Blue>))
123 {
124 // Start BLE but with paired device authorization only
125 bsp::ble::start_advertising(false);
126
127 bsp::lampda_print("Enable BLE for paired device");
128 }
129 }
130 // else: not bounded, or already started bounded connection, or a device is connected
131 else
132 {
133 // Start pairing mode. A connected device is disconnected, to force a new pairing
134 const bool isNotCurrentlyPairing = not(bsp::ble::is_advertising() and bsp::ble::is_open_to_all());
135 if (isNotCurrentlyPairing and
136 manager.overlay_animate_ramp(
137 holdDuration, 3000, modes::colors::PaletteGradient<modes::colors::Blue, modes::colors::Blue>))
138 {
139 // Start pairing mode
140 bsp::ble::force_pairing_mode();
141
142 bsp::lampda_print("Enable BLE pairing mode");
143 }
144 }
145 return true;
146 }
147 }
148
149 // sunset timer, on its configured clicks
150 if (logic::button_bindings::get_hold_behavior(clicks) == logic::button_bindings::Behavior::Sunset and
151 not isEndOfHoldEvent and holdDuration > 0)
152 {
153 auto manager = get_context();
154 if (manager.overlay_animate_ramp(
155 holdDuration, 1000, modes::colors::PaletteGradient<modes::colors::White, modes::colors::Red>))
156 {
157 // update the sunset timing
158 manager.state.isSunsetTimingPending = 2;
159 }
160
161 return true;
162 }
163
164 return false;
165}

238bool button_hold(const uint8_t clicks, const bool isEndOfHoldEvent, const uint32_t holdDuration)
239{
240 auto manager = get_context();
241 auto& rampHandler = manager.state.rampHandler;
242
243 // Add 5 minutes to sunset timer, on its configured clicks
244 if (logic::button_bindings::get_hold_behavior(clicks) == logic::button_bindings::Behavior::Sunset and
245 not isEndOfHoldEvent and holdDuration > 0 and logic::sunset_timer.is_enabled())
246 {
247 if (manager.overlay_animate_ramp(
248 holdDuration, 1000, modes::colors::PaletteGradient<modes::colors::White, modes::colors::Red>))
249 {
250 // update the sunset timing
251 manager.state.isSunsetTimingPending = 2;
252 }
253 }
254
255 // configure custom ramp, on its configured clicks. No ramps in favorite group
256 if (logic::button_bindings::get_hold_behavior(clicks) == logic::button_bindings::Behavior::CustomRamp and
257 not manager.state.isInFavoriteMockGroup)
258 {
259 rampHandler.update_ramp(manager.get_active_custom_ramp(), holdDuration, [&](uint8_t rampValue) {
260 manager.custom_ramp_update(rampValue);
261 manager.set_active_custom_ramp(rampValue);
262 });
263 return true;
264 }
265
266 switch (clicks)
267 {
268 case 13: // 13 clicks + hold: reset the stored parameters
269 {
270 if (not isEndOfHoldEvent and holdDuration > 0)
271 {
272 auto manager = get_context();
273 if (manager.overlay_animate_ramp(
274 holdDuration, 5000, modes::colors::PaletteGradient<modes::colors::Red, modes::colors::Red>))
275 {
276 bsp::lampda_print("clearing user parameters");
277
278 // reset the file system and memory
279 bsp::filesystem::user::format();
280 }
281 }
282 break;
283 }
284
285 case 20: // 20 clicks + hold: reset the whole system and stored parameters
286 {
287 if (not isEndOfHoldEvent and holdDuration > 0)
288 {
289 auto manager = get_context();
290 if (manager.overlay_animate_ramp(
291 holdDuration, 5000, modes::colors::PaletteGradient<modes::colors::Red, modes::colors::Red>))
292 {
293 // reset the file system and memory
294 bsp::lampda_print("clearing the whole file format");
295 bsp::filesystem::clear_internal_fs();
296
297 // shutdown the lamp
298 logic::behavior::internal::handle_shutdown_state();
299 }
300 }
301 break;
302 }
303 }
304
305 return false;
306}

487 manager.set_active_mode(manager.get_modes_count() - 1);
488 return;
489 }
490
491 manager.blip(250);

496bool handle_user_command(const common::UserCommand& command)
497{
498 switch (command.get_type())
499 {
500 case common::UserCommand::Type::SetUserRamp:
501 {
502 uint8_t ramp;
503 if (command.parse_ramp(ramp))
504 __private::handle_ramp_command(ramp);
505 else
506 bsp::lampda_print("Failed to parse set_user_ramp command");
507 return true;
508 }
509 case common::UserCommand::Type::Brightness:
510 {
511 brightness_t brgt;
512 if (command.parse_brightness(brgt))
513 __private::handle_brightness_control(brgt);
514 else
515 bsp::lampda_print("Failed to parse brightness command");
516 return true;
517 }
518 case common::UserCommand::Type::SetMode:
519 {
520 uint8_t groupId;
521 uint8_t modeId;
522 if (command.parse_set_mode(groupId, modeId))
523 __private::handle_mode_control(groupId, modeId);
524 else
525 bsp::lampda_print("Failed to parse set_mode command");
526 return true;
527 }
528 case common::UserCommand::Type::OnOff:
529 {
530 bool shouldTurnOn;
531 if (command.parse_turn_onoff(shouldTurnOn))
532 __private::handle_on_off_command(shouldTurnOn);
533 else
534 bsp::lampda_print("Failed to parse onoff command");
535 return true;
536 }
537 case common::UserCommand::Type::SetRealTime:
538 {
539 component::time::RealTime time;
540 if (command.parse_set_real_time_command(time))
541 __private::handle_set_time_command(time);
542 else
543 bsp::lampda_print("Failed to parse set_real_time command");
544 return true;
545 }
546 case common::UserCommand::Type::SetSunsetToTime:
547 {
548 component::time::RealTime time;
549 if (command.parse_set_sunset_to_time_command(time))
550 __private::handle_sunset_to_time_command(time);
551 else
552 bsp::lampda_print("Failed to parse set_sunset_time command");
553 return true;
554 }
555 case common::UserCommand::Type::SetSunriseToTime:
556 {
557 component::time::RealTime time;
558 if (command.parse_set_sunrise_to_time_command(time))
559 __private::handle_sunrise_to_time_command(time);
560 else
561 bsp::lampda_print("Failed to parse set_sunrise_time command");
562 return true;
563 }
564
565 case common::UserCommand::Type::SignalSunsetProgress:
566 {
567 float progress;
568 if (command.parse_signal_sunset_progress_command(progress))
569 get_context().sunset_update(progress);
570 else
571 bsp::lampda_print("Failed to parse signal_sunset_progress command");
572 return true;
573 }
574 case common::UserCommand::Type::SignalSunriseProgress:
575 {
576 float progress;
577 if (command.parse_signal_sunrise_progress_command(progress))
578 get_context().sunrise_update(progress);
579 else
580 bsp::lampda_print("Failed to parse signal_sunrise_progress command");
581 return true;
582 }
583 case common::UserCommand::Type::SignalSuntimerCancel:
584 {
585 if (command.parse_signal_suntimer_cancel_command())
586 {
587 auto manager = get_context();
588 manager.sunset_timer_internal_reset();
589 manager.sunrise_timer_internal_reset();
590 }
591 else
592 bsp::lampda_print("Failed to parse signal_suntimer_cancel command");
593 return true;
594 }
595 case common::UserCommand::Type::DisableSunTimer:
596 {
597 bool isSunrise;
598 if (command.parse_disable_sun_timer_command(isSunrise))
599 {
600 if (isSunrise)
601 {
602 // keep the brightness reached by the sunrise, as for a user brightness change
603 logic::sunrise_timer.cancel_at_current_brightness();
604 bsp::lampda_print("Sunrise timer disabled");
605 }
606 else
607 {
608 logic::sunset_timer.cancel();
609 bsp::lampda_print("Sunset timer disabled");
610 }
611 }
612 else
613 bsp::lampda_print("Failed to parse disable_sun_timer command");
614 return true;
615 }
616 case common::UserCommand::Type::SetMicSensitivity:
617 {
618 uint8_t sensitivity;
619 if (command.parse_set_mic_sensitivity_command(sensitivity))
620 component::microphone::set_sensitivity(sensitivity);
621 else
622 bsp::lampda_print("Failed to parse set_mic_sensitivity command");
623 return true;
624 }
625 case common::UserCommand::Type::BlipOutput:
626 {
627 uint32_t duration_ms;
628 if (command.parse_blip_output_command(duration_ms))
629 {
630 auto manager = get_context();
631 // skipped if a blip is running: it would replace its remaining time
632 if (manager.state.skipNextFrameEffect == 0)
633 manager.blip(duration_ms);
634 }
635 else
636 bsp::lampda_print("Failed to parse blip_output command");
637 return true;
638 }
639 case common::UserCommand::Type::PrintStatus:
640 {
641 if (command.parse_print_status_command())
642 {
643 if (not logic::behavior::is_in_output_state())
644 {
645 bsp::lampda_print_raw("$stat:OFF\r\n");
646 }
647 else
648 {
649 auto manager = get_context();
650 // single line, easy to parse by an external app: $stat:<group>:<mode>:<ramp>:<brightness>:<F|->
651 bsp::lampda_print_raw("$stat:g%u:m%u:r%u:%c brgt:%u\r\n",
652 manager.get_active_group(),
653 manager.get_active_mode(),
654 manager.get_active_custom_ramp(),
655 manager.state.isInFavoriteMockGroup ? 'F' : 'X',
656 logic::brightness::get_brightness());
657 }
658 }
659 else
660 bsp::lampda_print("Failed to parse print_status command");
661 return true;
662 }
663 case common::UserCommand::Type::PrintFavorites:
664 {
665 if (command.parse_print_favorites_command())
666 {
667 auto manager = get_context();
668 bsp::lampda_print("favorites set: %u (max %u), in favorite group: %s",
669 manager.state.usedFavoriteCount,
670 manager.state.maxFavoriteCount,
671 manager.state.isInFavoriteMockGroup ? "true" : "false");
672 }
673 else
674 bsp::lampda_print("Failed to parse print_favorites command");
675 return true;
676 }
677
678 default:
679 break;
680 }
681 return false;
682}
