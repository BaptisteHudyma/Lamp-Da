51inline size_t format(char* buffer, const char* name, const uint32_t* values, const size_t valueCount)
52{
53 int len = snprintf(buffer, maxFormatLenght, "$%s", name);
54 for (size_t i = 0; i < valueCount and len > 0 and static_cast<size_t>(len) < maxFormatLenght; ++i)
55 len += snprintf(buffer + len, maxFormatLenght - len, ":%lu", static_cast<unsigned long>(values[i]));
56
57 // snprintf returns the wanted lenght, so truncation shows as len >= buffer size
58 if (len <= 0 or static_cast<size_t>(len) >= maxFormatLenght)
59 {
60 buffer[0] = '\0';
61 return 0;
62 }
63 return static_cast<size_t>(len);
64}

119inline bool same_as_sent(const Slot& s)
120{
121 if (not s.hasSent)
122 return false;
123 for (size_t i = 0; i < s.count; ++i)
124 if (s.sent[i] != s.pendingValues[i])
125 return false;
126 return true;
127}

130inline void flush(Slot& s, const uint32_t nowMs)
131{
132 if (not s.pending)
133 return;
134
135 // the latest value is what the receiver already has: nothing to send
136 if (s.dedupe and not s.force and same_as_sent(s))
137 {
138 s.pending = false;
139 return;
140 }
141
142 // unsigned subtraction is safe with counter wrap around
143 if (s.hasSent and (nowMs - s.sentTimeMs) < signalIntervalMs)
144 return;
145
146 send(s.name, s.pendingValues, s.count);
147 for (size_t i = 0; i < s.count; ++i)
148 s.sent[i] = s.pendingValues[i];
149 s.hasSent = true;
150 s.sentTimeMs = nowMs;
151 s.pending = false;
152 s.force = false;
153}

156inline void submit(const SlotId id, const uint32_t* values, const bool forceSend = false)
157{
158 Slot& s = slot(id);
159 for (size_t i = 0; i < s.count; ++i)
160 s.pendingValues[i] = values[i];
161 s.pending = true;
162 s.force = s.force or forceSend;
163 flush(s, hal::time_ms());
164}
