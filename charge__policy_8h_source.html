62inline bool is_over_voltage(const Inputs& in)
63{
64 return in.isBalancerValid and (in.maxCell_mV > cellOverVoltage_mV or in.pack_mV > packOverVoltage_mV);

68inline bool is_end_of_charge(const Inputs& in)
69{
70 if (not in.isCharging)
71 return false;
72 // pack voltage reached the target
73 if (in.isBalancerValid and in.pack_mV >= physical_constants::batteryMaxVoltageSafe_mV)
74 return true;
75 // constant voltage phase, current tapered off.
76 // The level condition prevents a weak power source (low charge current) to be mistaken for an end of charge
77 return in.level >= fullLevel and in.chargeCurrent_mA <= physical_constants::batteryEndOfChargeCurrent_mA;
78}

90 bool update(const uint32_t time_ms, const Inputs& in)
91 {
92 // safety: over voltage stops the charge, whatever the state
93 if (is_over_voltage(in))
94 {
95 _isOverVoltage = true;
96 _overVoltageTime_ms = time_ms;
97 _isEndOfChargeDeglitching = false;
98 // already full: end the charge
99 if (in.level >= fullLevel)
100 _isFull = true;
101 return false;
102 }
103
104 // stay stopped until the cells are back under the resume voltage
105 if (_isOverVoltage)
106 {
107 if (time_ms - _overVoltageTime_ms < overVoltagePause_ms or not in.isBalancerValid or
108 in.maxCell_mV >= cellResumeVoltage_mV)
109 return false;
110 _isOverVoltage = false;
111 }
112
113 // battery full: wait for the level to drop
114 if (_isFull)
115 {
116 if (in.level >= rechargeLevel)
117 return false;
118 _isFull = false;
119 }
120
121 if (is_end_of_charge(in))
122 {
123 if (not _isEndOfChargeDeglitching)
124 {
125 _isEndOfChargeDeglitching = true;
126 _endOfChargeTime_ms = time_ms;
127 }
128 else if (time_ms - _endOfChargeTime_ms > endOfChargeDeglitch_ms)
129 {
130 _isEndOfChargeDeglitching = false;
131 _isFull = true;
132 return false;
133 }
134 }
135 else
136 {
137 _isEndOfChargeDeglitching = false;
138 }
139 return true;
140 }
