71 utils::vec2d compute_speed_increment(const utils::vec3d& accelerationCartesian_m, const float deltaTime_s) const
72 {
73 // only the acceleration tangent to the cylinder surface moves the particle
74 const float tangentialAcceleration =
75 accelerationCartesian_m.y * fast_cos(theta_rad) - accelerationCartesian_m.x * fast_sin(theta_rad);
76
77 return utils::vec2d(aroundTubeRatio / cylinderRadius_m * tangentialAcceleration * deltaTime_s,
78 linearSpeedGain * accelerationCartesian_m.z * deltaTime_s);
79 }

87 utils::vec2d update_speed(const utils::vec3d& accelerationCartesian_m, const float deltaTime_s)
88 {
89 const auto& speedIncrement = compute_speed_increment(accelerationCartesian_m, deltaTime_s);
90 // start by dampening the speed
91 dampen_speed(speedDampening);
92
93 // update speed
94 thetaSpeed_radS =
95 lmpd_constrain<float>(thetaSpeed_radS + speedIncrement.x, -maxAngularSpeed_radS, maxAngularSpeed_radS);
96 zSpeed_mS = lmpd_constrain<float>(zSpeed_mS + speedIncrement.y, -maxVerticalSpeed_mmS, maxVerticalSpeed_mmS);
97
98 return utils::vec2d(speedIncrement.x * cylinderRadius_mm, speedIncrement.y * 1000.0f);
99 }

107 void apply_acceleration(const utils::vec3d& accelerationCartesian_m,
108 const float deltaTime_s,
109 const bool shouldContrain = true)
110 {
111 update_speed(accelerationCartesian_m, deltaTime_s);
112
113 static constexpr float angularUnit = LampTy::maxWidthFloat / c_TWO_PI;
114 static constexpr float verticalUnit = LampTy::ledStripWidth_mm * 1.5f;
115
116 // update position (limit speed to pixel unit per dt)
117 const float angularPositionIncrement =
118 lmpd_constrain<float>(thetaSpeed_radS * deltaTime_s, -angularUnit, angularUnit);
119 const float verticalPositionIncrement =
120 lmpd_constrain<float>(zSpeed_mS * deltaTime_s, -verticalUnit, verticalUnit) * 1000.0f;
121
122 // update particle position
123 z_mm += verticalPositionIncrement;
124 theta_rad += angularPositionIncrement;
125
126 // limit the derivation of the angle
127 theta_rad = wrap_angle(theta_rad);
128
129 // constrain to the lamp body
130 if (shouldContrain)
131 constraint_into_lamp_body();
132 else
133 _savedLampIndex = to_lamp_index_no_bounds();
134 }

154 void constraint_into_lamp_body()
155 {
156 _savedLampIndex = to_lamp_index_no_bounds();
157
158 // coordinates go from 0 to -max
159 if (z_mm > 0)
160 {
161 z_mm = 0;
162 zSpeed_mS = -zSpeed_mS * reboundCoeff;
163 _savedLampIndex = to_lamp_index_no_bounds();
164 }
165
166 if (z_mm < -LampTy::lampHeight_mm)
167 {
168 z_mm = -LampTy::lampHeight_mm;
169 zSpeed_mS = -zSpeed_mS * reboundCoeff;
170 _savedLampIndex = to_lamp_index_no_bounds();
171 }
172
173 // handle the real limits (led strip do not start and end at zero depth)
174 if (not is_led_index_valid(_savedLampIndex))
175 {
176 // we are too high above the first led
177 if (z_mm >= 0)
178 {
179 // go one unit lower
180 z_mm = -LampTy::ledStripWidth_mm * 1.5f;
181 zSpeed_mS = -zSpeed_mS * reboundCoeff;
182 }
183 else
184 {
185 // go one unit above
186 z_mm = -LampTy::lampHeight_mm + LampTy::ledStripWidth_mm * 1.5f;
187 zSpeed_mS = -zSpeed_mS * reboundCoeff;
188 }
189 // udpate index
190 _savedLampIndex = to_lamp_index_no_bounds();
191 }

216 static int16_t led_index_at(const float positionTheta_rad, const float positionZ_mm)
217 {
218 float turnFraction = positionTheta_rad * (1.0f / c_TWO_PI);
219 turnFraction -= fast_floor(turnFraction);
220 const float turn = fast_floor(-positionZ_mm * (1.0f / rowHeight_mm) - turnFraction + 0.5f);
221 return static_cast<int16_t>(fast_floor((turn + turnFraction) * ledsPerTurn + 0.5f));
222 }

236 static float potential_at_led(const utils::vec3d& accelerationCartesian_m, const int16_t ledIndex)
237 {
238 const float angle_rad = ledIndex * (c_TWO_PI / ledsPerTurn);
239 return -aroundTubeRatio * cylinderRadius_mm *
240 (accelerationCartesian_m.x * fast_cos(angle_rad) + accelerationCartesian_m.y * fast_sin(angle_rad)) -
241 accelerationCartesian_m.z * to_helix_z(ledIndex);

251 void move_by(const utils::vec2d& offset_mm, const int16_t ledIndex)
252 {
253 theta_rad += offset_mm.x * (1.0f / cylinderRadius_mm);
254 // offsets are small, so one turn is enough to stay in the 0 to 2*PI range
255 if (theta_rad < 0.0f)
256 theta_rad += c_TWO_PI;
257 else if (theta_rad >= c_TWO_PI)
258 theta_rad -= c_TWO_PI;
259 z_mm += offset_mm.y;
260 _savedLampIndex = ledIndex;
261 }

281 static float fast_sin(const float angle_rad)
282 {
283 float turns = angle_rad * (1.0f / c_TWO_PI);
284 turns -= fast_floor(turns + 0.5f);
285 return 8.0f * turns - 16.0f * turns * fabsf(turns);
286 }
