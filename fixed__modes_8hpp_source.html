19template<bool isStep = false, uint32_t rampUpdateLenght_ms = 55, bool shouldSaturateRamp = false> struct PaletteMode :
20 public modes::BasicMode
21{
22 static void on_enter_mode(auto& ctx)
23 {
24 // saturate the ramp
25 ctx.template set_config_bool<ConfigKeys::rampSaturates>(shouldSaturateRamp);
26 //
27 ctx.template set_config_bool<ConfigKeys::customRampAnimEffect>(false);
28 // this ramps should be slow
29 ctx.template set_config_u32<ConfigKeys::customRampStepSpeedMs>(rampUpdateLenght_ms);
30
31 // display a ramp
32 ctx.overlay_animate_ramp(255, ctx.state.palette, 1000);
33 }
34
35 static void loop(auto& ctx)
36 {
37 // clear previous animation
38 ctx.lamp.clear();
39
40 const uint8_t customRamp = ctx.get_active_custom_ramp();
41 uint32_t color = modes::colors::from_palette<not shouldSaturateRamp>(
42 // if palette is stepped, remove all intermediate colors
43 static_cast<uint8_t>(isStep ? (customRamp >> 4) << 4 : customRamp),
44 // not stepped, allow interpolation
45 ctx.state.palette);
46
47 // fill, with mask
48 ctx.lamp.fill(color);
49 }
