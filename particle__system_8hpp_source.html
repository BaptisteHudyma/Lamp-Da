148 void iterate_with_collisions(const utils::vec3d& accelerationCartesian,
149 const float deltaTime_s,
150 const bool shouldContrain = true,
151 const float viscosity = sandViscosity)
152 {
153 update_occupation();
154
155 frameIndex += 1;
156 const uint8_t flowReach = 1 + static_cast<uint8_t>((1.0f - viscosity) * maxFlowReach + 0.5f);
157
158 for (size_t i = 0; i < particuleCount; ++i)
159 {
160 // do not iterate non allocated particles
161 if (not isAllocated[i])
162 continue;
163
164 // spread the resting particles over the frames. A support that disappears is seen a few frames late, at worst
165 if (isResting[i] and ((i + frameIndex) % restingUpdatePeriod) != 0)
166 continue;
167
168 const utils::vec2d& drive = particules[i].update_speed(accelerationCartesian, deltaTime_s);
169 move_with_collisions(i, accelerationCartesian, drive, deltaTime_s, shouldContrain, flowReach);
170 }

179 template<typename ShouldDepopFunction> uint16_t depop_particules(const ShouldDepopFunction& shouldDepopFunction)
180 {
181 static_assert(std::is_invocable_r_v<bool, ShouldDepopFunction, Particle const&>);
182
183 uint16_t particleCount = 0;
184 for (size_t i = 0; i < particuleCount; ++i)
185 {
186 // do not depop non allocated particles
187 if (not isAllocated[i])
188 continue;
189
190 const auto& parti = particules[i];
191 if (shouldDepopFunction(parti))
192 {
193 isAllocated[i] = false;
194 if (parti._savedLampIndex >= 0 && parti._savedLampIndex < isOccupiedSpaces.size())
195 isOccupiedSpaces[parti._savedLampIndex] = false;
196 }
197 else
198 {
199 particleCount += 1;
200 }
201 }
202 return particleCount;

303 template<typename PositiongeneratorFunction>
304 void spawn_particule(const size_t index,
305 const PositiongeneratorFunction& positionGeneratorFunction,
306 const uint32_t color)
307 {
308 static_assert(std::is_invocable_r_v<int16_t, PositiongeneratorFunction, size_t>);
309
310 int16_t pos = positionGeneratorFunction(index);
311 int maxTries = 3;
312 while (ParticleSystem::is_position_taken(pos) and maxTries > 0)
313 {
314 pos = positionGeneratorFunction(index);
315 maxTries--;
316 }
317 // generate start position from user function
318 const auto& helixCoordinates = modes::strip_to_helix_unconstraint(pos);
319 particules[index] = Particle(utils::vec3d {helixCoordinates.x, helixCoordinates.y, helixCoordinates.z});
320 particules[index].set_color(color);
321 if (index < isAllocated.size())
322 {
323 isAllocated[index] = true;
324 isResting[index] = false;
325 }
326 const int16_t ledIndex = particules[index]._savedLampIndex;
327 if (ledIndex >= 0 && ledIndex < isOccupiedSpaces.size())
328 isOccupiedSpaces[ledIndex] = true;
329 }

361 void move_particle(const size_t slot, const utils::vec2d& offset_mm, const int16_t toIndex)
362 {
363 Particle& particle = particules[slot];
364 const int16_t fromIndex = particle._savedLampIndex;
365 if (toIndex != fromIndex)

386 void move_with_collisions(const size_t slot,
387 const utils::vec3d& acceleration,
388 const utils::vec2d& drive,
389 const float deltaTime_s,
390 const bool shouldContrain,
391 const uint8_t flowReach)
392 {
393 Particle& particle = particules[slot];
394
395 utils::vec2d step = particle.get_surface_speed().multiply(deltaTime_s);
396 const float longestAxis_mm = std::max(fabsf(step.x), fabsf(step.y));
397 const float maxStep_mm = maxCollisionSpeed_mmS * deltaTime_s;
398 if (longestAxis_mm > maxStep_mm)
399 step = step.multiply(maxStep_mm / longestAxis_mm);
400
401 const int16_t fromIndex = particle._savedLampIndex;
402 const int16_t toIndex = particle.led_index_at_offset(step);
403 if (toIndex == fromIndex or not is_led_blocked(fromIndex, toIndex, shouldContrain))
404 {
405 move_particle(slot, step, toIndex);
406 isResting[slot] = false;
407 return;
408 }
409
410 particle.dampen_speed(contactDampening);
411 // only the acceleration can make a particle flow, or the rebounds would agitate the piles forever
412 const bool hasFlowed =
413 drive.dot(step) > 0.0f and flow_around_obstacle(slot, acceleration, drive, shouldContrain, flowReach);
414 isResting[slot] = not hasFlowed;
415 }

428 bool flow_around_obstacle(const size_t slot,
429 const utils::vec3d& acceleration,
430 const utils::vec2d& drive,
431 const bool shouldContrain,
432 const uint8_t flowReach)
433 {
434 const Particle& particle = particules[slot];
435 const int16_t fromIndex = particle._savedLampIndex;
436
437 const utils::vec2d& downhill = drive.multiply(1.0f / sqrtf(drive.dot(drive)));
438 const utils::vec2d& oneLedDown = downhill.multiply(led_span_along(downhill));
439 const utils::vec2d sideDirection(-downhill.y, downhill.x);
440 const utils::vec2d& oneLedAside = sideDirection.multiply(led_span_along(sideDirection));
441
442 // no side preference, or the piles would lean over time
443 const float firstSide = (random8() & 1) ? 1.0f : -1.0f;
444 for (uint8_t distance = 1; distance <= flowReach; ++distance)
445 {
446 // only the surface flows far: it spares the scans for the many particles buried in a pile
447 if (distance == 2 and
448 is_led_blocked(fromIndex, particle.led_index_at_offset(oneLedDown.multiply(-1.0f)), shouldContrain))
449 return false;
450
451 for (const float side: {firstSide, -firstSide})
452 {
453 const utils::vec2d& gapOffset = oneLedAside.multiply(side * distance).add(oneLedDown);
454 const int16_t gapIndex = particle.led_index_at_offset(gapOffset);
455 if (gapIndex != fromIndex and not is_led_blocked(fromIndex, gapIndex, shouldContrain) and
456 is_flow_worth_it(acceleration, fromIndex, gapIndex))
457 {
458 move_particle(slot, gapOffset, gapIndex);
459 return true;
460 }
461 }
462 }
463 return false;
464 }
465
468 static bool is_flow_worth_it(const utils::vec3d& acceleration, const int16_t fromIndex, const int16_t toIndex)
469 {
470 const float minDrop = minFlowDropRatio * rowHeight_mm * sqrtf(acceleration.dot(acceleration));
471 return Particle::potential_at_led(acceleration, toIndex) <
472 Particle::potential_at_led(acceleration, fromIndex) - minDrop;
473 }
474
476 static float led_span_along(const utils::vec2d& direction)
477 {
478 return fabsf(direction.x) * ledLength_mm + fabsf(direction.y) * rowHeight_mm;
479 }
