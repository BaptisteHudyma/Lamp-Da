88template<typename UIntTy = uint8_t>
89static constexpr uint32_t blend(uint32_t leftColor, uint32_t rightColor, UIntTy blendIndex)
90{
91 static_assert(std::is_same_v<UIntTy, uint8_t> || std::is_same_v<UIntTy, uint16_t>, "u8 or u16 allowed only");
92 constexpr UIntTy blendmax = (std::is_same_v<UIntTy, uint16_t> ? UINT16_MAX : UINT8_MAX);
93
94 ToRGB left_rgb(leftColor);
95 ToRGB right_rgb(rightColor);
96 // left + (right - left) * (blendIndex / maxBlend)
97 // Optimized to
98 // (left * (maxBlend - blendIndex) + right * blendIndex) / maxBlend
99 return fromRGB((static_cast<uint32_t>(left_rgb.r) * (blendmax - blendIndex) +
100 static_cast<uint32_t>(right_rgb.r) * blendIndex) /
101 blendmax,
102 (static_cast<uint32_t>(left_rgb.g) * (blendmax - blendIndex) +
103 static_cast<uint32_t>(right_rgb.g) * blendIndex) /
104 blendmax,
105 (static_cast<uint32_t>(left_rgb.b) * (blendmax - blendIndex) +
106 static_cast<uint32_t>(right_rgb.b) * blendIndex) /
107 blendmax);
108}

121template<bool isVideoMode = false> static uint32_t fade(uint32_t inputColor, uint8_t fadeAmount)
122{
123 uint32_t res;
124 ToRGB input_rgb(inputColor);
125
126 if (isVideoMode)
127 {
128 res = fromRGB(scale8_video(input_rgb.r, fadeAmount),
129 scale8_video(input_rgb.g, fadeAmount),
130 scale8_video(input_rgb.b, fadeAmount));
131 }
132 else
133 {
134 res = fromRGB(scale8(input_rgb.r, fadeAmount), scale8(input_rgb.g, fadeAmount), scale8(input_rgb.b, fadeAmount));
135 }
136 return res;
137}

140template<bool isFast = false> uint32_t add(uint32_t c1, uint32_t c2)
141{
142 ToRGB input1_rgb(c1);
143 ToRGB input2_rgb(c2);
144
145 if (isFast)
146 {
147 return fromRGB(
148 qadd8(input1_rgb.r, input2_rgb.r), qadd8(input1_rgb.g, input2_rgb.g), qadd8(input1_rgb.b, input2_rgb.b));
149 }
150 else
151 {
152 uint32_t r = input1_rgb.r + input2_rgb.r;
153 uint32_t g = input1_rgb.g + input2_rgb.g;
154 uint32_t b = input1_rgb.b + input2_rgb.b;
155 uint16_t max = r;
156 if (g > max)
157 max = g;
158 if (b > max)
159 max = b;
160 if (max < 256)
161 {
162 return fromRGB(r, g, b);
163 }
164 else
165 {
166 return fromRGB(r * 255 / max, g * 255 / max, b * 255 / max);
167 }
168 }
