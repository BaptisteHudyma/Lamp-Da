118 void write_to_led_driver(const uint8_t writeBrightness)
119 {
120 // Adjust brightness to the desired output
121 const uint8_t capedShown = shownCount % refreshFramesCount;
122 for (uint16_t i = 0; i < LED_COUNT; ++i)
123 {
124 // ignore masked leds
125 if (nextWriteMask[i])
126 {
127 // write black
128 StripImpl_t::setPixelColor(i, 0);
129
130 if constexpr (useTemporalDithering)
131 {
132 _colorErrors[i] = 0;
133 }
134 // and skip
135 continue;
136 }
137
138 if constexpr (useTemporalDithering)
139 {
140 const uint32_t c = convert_color_with_brightness(_colors[i], writeBrightness, i + capedShown, _colorErrors[i]);
141 // set strip color
142 StripImpl_t::setPixelColor(i, c);
143 }
144 else
145 {
146 const uint32_t c = convert_color_with_brightness(_colors[i], writeBrightness, i + capedShown, _colorErrors[0]);
147 // set strip color
148 StripImpl_t::setPixelColor(i, c);
149 }
150 }
151
152 // reset the mask after write
153 nextWriteMask.reset();
154 }

181 static uint8_t restore_color_with_brightness(uint8_t colorIn, const uint8_t brightness)
182 {
183 // only special case
184 const uint16_t colorShifted = static_cast<uint16_t>(colorIn) << 8u;
185 if (colorShifted <= brightness or brightness == 0)
186 return 0;
187
188 uint8_t fullColor = static_cast<uint8_t>((colorShifted - brightness) / brightness);
189 return fullColor;
190 }

200 static std::pair<uint8_t, uint8_t> get_brightness_color_and_error(const uint8_t colorIn,
201 uint16_t errorIn,
202 const uint8_t brightness,
203 const uint16_t index)
204 {
205 // blue noise look up table
206 static constexpr std::array<uint8_t, 64> BLUE_NOISE_LUT = {
207 0, 32, 8, 40, 2, 34, 10, 42, 48, 16, 56, 24, 50, 18, 58, 26, 12, 44, 4, 36, 14, 46,
208 6, 38, 60, 28, 52, 20, 62, 30, 54, 22, 3, 35, 11, 43, 1, 33, 9, 41, 51, 19, 59, 27,
209 49, 17, 57, 25, 15, 47, 7, 39, 13, 45, 5, 37, 63, 31, 55, 23, 61, 29, 53, 21};
210
211 // only special case
212 if (colorIn == 0)
213 return {0, 0};
214
215 // scale color by brightness
216 const uint16_t scaledColor = colorIn * brightness + brightness;
217
218 // When using standard and temporal dithering, shift the pattern
219 if constexpr (useColorDithering and useTemporalDithering)
220 {
221 const uint8_t allowedError = UINT8_MAX - (scaledColor & 0xFF);
222 errorIn += min<uint16_t>(allowedError, BLUE_NOISE_LUT[index % BLUE_NOISE_LUT.size()]);
223 }
224
225 // add error
226 const uint32_t fullColor = min<uint32_t>(scaledColor + errorIn, 0xFF00);
227
228 // compute final color and new error
229 uint32_t finalColorRaw = fullColor >> 8;
230 const uint8_t finalError = fullColor & 0xFF;
231
232 // When using only color dithering, add a blue noise error to the final color
233 if constexpr (useColorDithering and not useTemporalDithering)
234 {
235 finalColorRaw += (finalError > BLUE_NOISE_LUT[index % BLUE_NOISE_LUT.size()] ? 1 : 0);
236 }
237
238 const uint8_t finalColor = (finalColorRaw > 255) ? 255 : static_cast<uint8_t>(finalColorRaw);
239 return {finalColor, finalError};
240 }

250 static uint32_t convert_color_with_brightness(const uint32_t color,
251 const uint8_t brightness,
252 const uint16_t index,
253 uint32_t& error)
254 {
255 // no temporal dithering: no error propagation
256 if constexpr (not useTemporalDithering)
257 {
258 error = 0;
259 }
260
261 utils::ColorSpace::RGB c(color);
262 utils::ColorSpace::RGB err(error);
263
264 // convert colors and errors, with small offsets for the different chanels
265 const auto& [red, redError] = get_brightness_color_and_error(c.get_rgb().red, err.get_rgb().red, brightness, index);
266 const auto& [green, greenError] =
267 get_brightness_color_and_error(c.get_rgb().green, err.get_rgb().green, brightness, index + 1);
268 const auto& [blue, blueError] =
269 get_brightness_color_and_error(c.get_rgb().blue, err.get_rgb().blue, brightness, index + 2);
270
271 // store error components
272 error = utils::ColorSpace::RGB(redError, greenError, blueError).get_rgb().color;
273 return utils::ColorSpace::RGB(red, green, blue).get_rgb().color;
274 }
