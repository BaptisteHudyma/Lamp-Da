45 void loop(auto& ctx)
46 {
47 if (isComplete)
48 return;
49
50 spawn_pending(ctx);
51
52 // mask all the leds that needs masking
53 for (uint16_t i = 0; i < maskedLeds.size(); i++)
54 {
55 if (maskedLeds[i])
56 ctx.lamp.mask_led_at_index(i, true);
57 }
58
59 if (particuleSystem.get_number_of_active() > 0)
60 {
61 static constexpr bool shouldKeepInLampBounds = false;
62 particuleSystem.iterate_no_collisions(
63 utils::vec3d(0.0, 0.0, -9.81 / 4.0), ctx.lamp.frameDurationMs / 1000.0, shouldKeepInLampBounds);
64
65 // unmask and remove the particles that reached their target
66 const uint16_t left = particuleSystem.depop_particules_indexed([&](size_t slot, const Particle& p) {
67 return land_if_arrived(ctx, slot, p);
68 });
69
70 // Force display because the pixels below are masked
71 static constexpr bool forceDisplayParticles = true;
72 particuleSystem.show(ctx.lamp, forceDisplayParticles);
73
74 if (left == 0 and spawned >= ctx.lamp.ledCount)
75 isComplete = true;
76 }
77 }

83 void update_progress_rate(auto& ctx, const float expectedProgress)
84 {
85 if (isComplete)
86 return;
87
88 const float progress = lmpd_constrain<float>(expectedProgress, 0.0f, 1.0f);
89
90 // done: reveal everything, whatever is still falling
91 if (progress >= 1.0f)
92 {
93 complete(ctx);
94 return;
95 }
96
97 if (progress == latestProgress)
98 return;
99
100 if (latestProgress >= 0.0f and ctx.lamp.now > latestProgressTime)
101 {
102 const float elapsedMs = static_cast<float>(ctx.lamp.now - latestProgressTime);
103 progressPerMs = lmpd_constrain<float>((progress - latestProgress) / elapsedMs, 0.0f, 1.0f);
104 }
105 else
106 {
107 progressPerMs = 0.0f; // first call: rate unknown, only catch up to the current progress
108 }
109 latestProgress = progress;
110 latestProgressTime = ctx.lamp.now;
111 }

113protected:
115 float estimated_progress(auto& ctx) const
116 {
117 if (latestProgress < 0.0f)
118 return 0.0f;
119 const float elapsedMs = static_cast<float>(ctx.lamp.now - latestProgressTime) + fallLeadMs;
120 return lmpd_constrain<float>(latestProgress + progressPerMs * elapsedMs, latestProgress, 1.0f);
121 }

134 void spawn_pending(auto& ctx)
135 {
136 const uint16_t total = ctx.lamp.ledCount;
137 const uint16_t wanted = static_cast<uint16_t>(estimated_progress(ctx) * total + 0.5f);
138
139 // limit the spawns per frame: no burst, particles are spread over time
140 for (uint8_t i = 0; i < maxSpawnPerLoop and spawned < wanted; i++)
141 {
142 if (not spawn_one(ctx))
143 break; // no free slot or no free target: retry next frame
144 }
145 }

147 bool spawn_one(auto& ctx)
148 {
149 if (particuleSystem.get_number_of_active() >= MaxParticles)
150 return false;
151
152 uint32_t targetId;
153 if (not pick_target(ctx, targetId))
154 return false;
155
156 // spawn at the top of the lamp, same angle as the target
157 const auto helix = modes::strip_to_helix_unconstraint(static_cast<int16_t>(targetId));
158 const float theta = atan2(helix.y, helix.x);
159 const int16_t spawnIndex = modes::to_led_index_no_bounds(theta, 0.0f);
160 const float targetZ = helix.z;
161 const uint32_t color = ctx.lamp.getPixelColor(targetId);
162
163 const bool notAllCreated = particuleSystem.init_deferred_particules(
164 1,
165 [this, targetId, targetZ, spawnIndex](size_t slot) -> int16_t {
166 this->targets[slot] = targetId;
167 this->targetsZ[slot] = targetZ;
168 return spawnIndex;
169 },
170 [color](size_t) {
171 return color;
172 });
173 if (notAllCreated)
174 return false; // no free slot
175
176 reserved.set(targetId);
177 spawned += 1;
178 return true;
179 }

182 bool is_candidate(auto& ctx, uint16_t x, int16_t y)
183 {
184 const auto id = modes::to_strip(x, y);
185 if (id < 0 or id >= ctx.lamp.ledCount)
186 return false;
187 return maskedLeds[id] and not reserved[id];
188 }
189
191 bool pick_target(auto& ctx, uint32_t& targetId)
192 {
193 for (int16_t y = ctx.lamp.maxHeight; y >= 0; y--)
194 {
195 uint16_t count = 0;
196 for (uint16_t x = 0; x <= ctx.lamp.maxWidth; x++)
197 {
198 if (is_candidate(ctx, x, y))
199 count += 1;
200 }
201 if (count == 0)
202 continue;
203
204 uint16_t selected = static_cast<uint16_t>(rand() % count);
205 for (uint16_t x = 0; x <= ctx.lamp.maxWidth; x++)
206 {
207 if (not is_candidate(ctx, x, y))
208 continue;
209 if (selected == 0)
210 {
211 targetId = static_cast<uint32_t>(modes::to_strip(x, y));
212 return true;
213 }
214 selected -= 1;
215 }
216 }
217 return false; // every pixel is revealed or targeted
218 }

221 bool land_if_arrived(auto& ctx, const size_t slot, const Particle& p)
222 {
223 // z goes from 0 (top) to -lampHeight (bottom)
224 if (p.z_mm > targetsZ[slot])
225 return false;
226
227 const uint32_t target = targets[slot];
228 maskedLeds.reset(target); // visible
229 reserved.reset(target);
230 return true;
231 }
